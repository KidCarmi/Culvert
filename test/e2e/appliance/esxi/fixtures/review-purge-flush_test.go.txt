package badger

import (
 "bytes"
 "fmt"
 "os"
 "path/filepath"
 "strings"
 "testing"
 "time"
)
type reviewFlushLogger struct{ failed chan struct{} }
func (l *reviewFlushLogger) Errorf(f string,a ...interface{}) { if strings.Contains(fmt.Sprintf(f,a...),"error flushing memtable") { select { case l.failed<-struct{}{}: default: } } }
func (*reviewFlushLogger) Warningf(string,...interface{}) {}
func (*reviewFlushLogger) Infof(string,...interface{}) {}
func (*reviewFlushLogger) Debugf(string,...interface{}) {}
func TestReviewPurgeAfterFlushFailure(t *testing.T) {
 dir:=t.TempDir()
 logger:=&reviewFlushLogger{failed:make(chan struct{},1)}
 db,err:=Open(DefaultOptions(dir).WithLogger(logger).WithMemTableSize(1<<20).WithValueLogFileSize(1<<20).WithValueThreshold(1024).WithNumCompactors(0).WithCompactL0OnClose(false))
 if err!=nil {t.Fatal(err)}
 defer func(){_ = db.Close(); for _,mt:=range db.imm {if mt.wal!=nil { _=mt.wal.Close(-1) }} }()
 for i:=1;i<=50;i++ {if err:=os.Mkdir(filepath.Join(dir,fmt.Sprintf("%06d.sst",i)),0700);err!=nil {t.Fatal(err)}}
 put:=func(i int){t.Helper();if err:=db.Update(func(txn *Txn)error{return txn.Set([]byte(fmt.Sprintf("key%08d",i)),bytes.Repeat([]byte{0x41},512))});err!=nil {t.Fatal(err)}}
 for i:=0;i<2000;i++ { put(i) }
 select{case <-logger.failed: case <-time.After(5*time.Second):t.Fatal("no genuine SST creation failure")}
 t.Log("observed actual SST creation error before calling public DropAll")
 if err:=db.DropAll();err!=nil {t.Fatal(err)}
 db.lock.RLock();afterPurge:=len(db.imm);db.lock.RUnlock()
 if afterPurge!=0 {t.Fatalf("DropAll did not clear old imm: %d",afterPurge)}
 t.Log("DropAll returned nil and cleared every pre-purge immutable memtable")
 for i:=1;i<=50;i++ {if err:=os.Remove(filepath.Join(dir,fmt.Sprintf("%06d.sst",i)));err!=nil {t.Fatal(err)}}
 t.Log("removed all 50 SST filename blockers before subsequent writes")
 for i:=0;i<15000;i++ {put(i)}
 time.Sleep(time.Second)
 db.lock.RLock(); retained:=len(db.imm);db.lock.RUnlock()
 t.Logf("after 1s drain wait: retained=%d NumMemtables=%d tables=%d queued=%d",retained,db.opt.NumMemtables,len(db.Tables()),len(db.flushChan))
 if retained>db.opt.NumMemtables && len(db.Tables())==0 {t.Fatal("resumed flusher remains poisoned: unbounded retained memtables and no SST")}
}
